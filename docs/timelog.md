# Автоматизация

```text
#!/bin/bash
# scripts/log-time.sh
set -euo pipefail

FILE="docs/TIMELOG.md"
DATE=$(date +%Y-%m-%d)

read -p "Номер шага (например, 2.1): " STEP
read -p "Оценка (часы): " EST

echo "| $DATE | $STEP | $EST | | | |" >> "$FILE"
echo "Записано в $FILE"
```

```
# Средняя точность оценки
awk -F'|' 'NR>3 && $4!="" {
  est=$4; fact=$5;
  if (est>0) dev=(fact-est)/est*100;
  sum+=dev; n++
} END {print "Среднее отклонение:", sum/n "%"}' docs/TIMELOG.md
```
